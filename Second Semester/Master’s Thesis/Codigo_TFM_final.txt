# FASE 2: Control de Calidad y Trimming

## Procesamiento de lecturas crudas (Illumina y Nanopore).

## 2.1. Crear los directorios que necesarios para hacer el analisis de qualidad y trimming de las secuencias
mkdir -p quality/raw quality/filtered trimmed

## 2.2. Generación de la lista de Muestras
fastqc *fastq.gz -o quality/raw/
ls *fastq.gz | cut -d _ -f 2 | sort -u > muestras.txt # (solo muestras de illumina - se hizo en dos tiempos diferentes)
ls *fastq.gz | cut -d _ -f 2 | sort -u > muestras_2.txt # (solo muestras de nanopore - se hizo en dos tiempos diferentes)

## 2.3. Análisis de la calidad inicial

### 2.3.1. Análisis de fastqc para las reads
fastqc --memory 8192 *fastq.gz -o quality/raw/

### 2.3.2. Integración de los resultados de calidad
multiqc .

## 2.4. Trimming (fastp para Illumina, NanoFilt para Nanopore)

### 2.4.1. Illumina
for i in $(cat muestras.txt); do 
    fastp --in1 *$i*_1* --in2 *$i*_2* \
          --out1 trimmed/${i}_R1_filtered.fastq.gz --out2 trimmed/${i}_R2_filtered.fastq.gz \
          --detect_adapter_for_pe --cut_front --cut_tail --cut_mean_quality 30 \
          --json trimmed/$i.json --html trimmed/$i.html
done

### 2.4.2. Nanopore
for i in $(cat muestras_2.txt); do 
    gunzip -c ${i}.fastq.gz | NanoFilt -q 10 -l 500 | gzip > trimmed/${i}_filtered.fastq.gz
done

### 2.5. Análisis de la calidad después del trimming
fastqc --memory 8192 trimmed/*fastq.gz -o quality/filtered

## 2.6. Integración de los resultados de calidad
multiqc .
